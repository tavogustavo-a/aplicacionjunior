``````````````````````````````````````````````````````````
## 5. Servicio systemd
## Puerto 8001 a propósito: el 8000 ya lo usa el otro sitio de tuyaho.com.
## Un solo worker: la base SQLite no se comparte entre varios procesos.
## El código está en utilidades/; gunicorn se arranca desde ahí.
sudo tee /etc/systemd/system/aplicacionjunior.service > /dev/null <<EOF
[Unit]
Description=aplicacionjunior
After=network.target

[Service]
User=junior
Group=junior
WorkingDirectory=/home/junior/aplicacionjunior/utilidades
Environment="PATH=/home/junior/aplicacionjunior/venv/bin"
Environment="BEHIND_PROXY=1"
ExecStart=/home/junior/aplicacionjunior/venv/bin/gunicorn --workers 1 --bind 127.0.0.1:8001 --timeout 3600 --forwarded-allow-ips=127.0.0.1 app:app
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF
``````````````````````````````````````````````````````````
sudo systemctl daemon-reload
sudo systemctl start aplicacionjunior
sudo systemctl enable aplicacionjunior
sudo systemctl status aplicacionjunior
``````````````````````````````````````````````````````````
## 6. Permisos del home para que nginx pueda llegar después, sin leer el .env
chmod 711 /home/junior
sudo usermod -aG junior www-data
``````````````````````````````````````````````````````````
## 7. Comprobar que la app responde dentro del servidor
curl -I http://127.0.0.1:8001/login
``````````````````````````````````````````````````````````
## 8. Nginx solo para junior.tuyaho.com
## Archivo nuevo. No abras pagina-monetizar.conf.
sudo tee /etc/nginx/sites-available/aplicacionjunior.conf > /dev/null <<'EOF'
server {
    listen 80;
    server_name junior.tuyaho.com;

    location / {
        proxy_pass http://127.0.0.1:8001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
        proxy_buffering off;
    }
}
EOF
sudo ln -sf /etc/nginx/sites-available/aplicacionjunior.conf /etc/nginx/sites-enabled/aplicacionjunior.conf
sudo nginx -t
sudo systemctl reload nginx
``````````````````````````````````````````````````````````
## 9. Certificado solo de este subdominio
## No pongas tuyaho.com en este comando.
sudo certbot --nginx -d junior.tuyaho.com --email admin@tuyaho.com --agree-tos --no-eff-email -n --redirect
sudo nginx -t
sudo systemctl reload nginx
